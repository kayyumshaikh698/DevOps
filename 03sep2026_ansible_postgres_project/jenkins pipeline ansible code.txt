pipeline {
    agent any
    environment {
        host = 'ansible'
    }
    stages {
        stage('HOSTNAME') {
            steps {
                sh 'ssh -tq $host hostname -f'
            }
        }
        stage('IPADDRESS') {
            steps {
                sh 'ssh -tq $host hostname -i'
            }
        }
        stage('USER') {
            steps {
                sh 'ssh -tq $host whoami'
            }
        }
        stage('ansible') {
            steps {
                // Connects directly using SSH keys as user kayyum
                sh 'ssh -tq kayyum@$host "ansible-playbook -i /home/kayyum/inventory.ini /home/kayyum/playbook.yml"'
            }
        }
    }
}
