*** best query for Grafana for postgres  monitoring alert ****
SELECT
  NOW() AS time,
  datid,
  datname AS metric,
  numbackends AS active_sessions
FROM
  pg_catalog.pg_stat_database
WHERE
  datname IS NOT NULL 


SELECT
  NOW() AS time,
  datid,
  xact_commit as commits,
  datname AS metric
FROM
  pg_catalog.pg_stat_database
WHERE
  datname IS NOT NULL
